### Monitoring Implementation

```typescript
async function recordMetrics(
  metrics: ProductionMetrics,
  env: Env
): Promise<void> {
  // Store in D1 for historical analysis
  await env.DB.prepare(
    `INSERT INTO metrics (
      timestamp, p50_latency, p95_latency, p99_latency,
      requests_per_second, errors_per_second, success_rate,
      cache_hit_rate, db_latency, kv_latency, tokens_per_request
    ) VALUES (NOW(), ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
  ).bind(
    metrics.p50_latency_ms,
    metrics.p95_latency_ms,
    metrics.p99_latency_ms,
    metrics.requests_per_second,
    metrics.errors_per_second,
    metrics.success_rate,
    metrics.cache_hit_rate,
    metrics.db_latency_ms,
    metrics.kv_latency_ms,
    metrics.tokens_per_request
  ).run();

  // Alert if thresholds exceeded
  if (metrics.p95_latency_ms > 3000) {
    await alertTeam(
      "⚠️ High latency detected: " + metrics.p95_latency_ms + "ms",
      env
    );
  }

  if (metrics.success_rate < 0.95) {
    await alertTeam(
      "🔴 Low success rate: " + Math.round(metrics.success_rate * 100) + "%",
      env
    );
  }

  if (metrics.errors_per_second > 5) {
    await alertTeam(
      "🚨 High error rate: " + metrics.errors_per_second + " errors/sec",
      env
    );
  }
}

async function alertTeam(message: string, env: Env): Promise<void> {
  // Send to Slack
  await fetch(env.SLACK_WEBHOOK, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      text: message,
      channel: "#hermes-alerts",
    }),
  });

  // Send to PagerDuty if critical
  if (message.includes("🚨")) {
    await fetch("https://events.pagerduty.com/v2/enqueue", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        routing_key: env.PAGERDUTY_KEY,
        event_action: "trigger",
        payload: {
          summary: message,
          severity: "critical",
          source: "HERMES Trinity",
        },
      }),
    });
  }
}
```

## Scaling Strategy

### Automatic Scaling

Cloudflare Workers automatically scale based on demand, but configure these thresholds:

```toml
[limits]
cpu_ms = 30000  # 30 second CPU timeout
max_request_size_mb = 100

[env.production.limits]
# Aggressive limits to force optimization
cpu_ms = 10000  # 10 seconds
max_request_size_mb = 10

# Batch size per request
batch_size = 32
# If more: queue and process asynchronously
```

### Queue for High Load

When load exceeds capacity, queue requests:

```typescript
async function enqueueRequest(
  request: Request,
  env: Env
): Promise<Response> {
  const queue = env.QUEUE;
  const payload = await request.json();

  // For high-priority requests, process immediately
  if (payload.priority === "high") {
    return await processRequest(payload, env);
  }

  // For normal requests, queue if load is high
  const currentLoad = await getSystemLoad(env);
  if (currentLoad > 80) {
    await queue.send(payload);
    return new Response(
      JSON.stringify({
        status: "queued",
        requestId: payload.id,
        message: "Processing high load - request queued",
      }),
      { status: 202, headers: { "Content-Type": "application/json" } }
    );
  }

  // Low load - process immediately
  return await processRequest(payload, env);
}

async function getSystemLoad(env: Env): Promise<number> {
  const stats = await env.KV.get("system:stats", "json");
  return stats?.currentLoad || 0;
}
```

### Track Token Usage

```typescript
interface TokenBudget {

async function trackTokens(
  tokensUsed: number,
  userId: string,
  env: Env
): Promise<void> {
  // Update daily count
  const dailyKey = `tokens:daily:${new Date().toISOString().split("T")[0]}`;
  const daily = (await env.KV.get(dailyKey, "json")) || { total: 0 };
  daily.total += tokensUsed;
  await env.KV.put(dailyKey, JSON.stringify(daily), {
    expirationTtl: 86400, // 24 hours
  });

  // Check limits
  if (daily.total > 10000000) {
    // Stop accepting requests
    await alertTeam("🛑 Daily token limit reached", env);
  }
}
```

## Incident Response

### Response Playbook

1. **Detection** (Automated alerts)
   ```
   Alert → PagerDuty → On-call engineer
   ```

2. **Triage** (5 min)
   - Is it HERMES? Check `/status` endpoint
   - Is it infrastructure? Check Cloudflare dashboard
   - Is it KAI fallback? Check n8n webhooks

3. **Mitigation** (10 min)
   - Scale up Workers if CPU-bound
   - Clear KV cache if memory issue
   - Reduce batch size if latency high
   - Switch to KAI fallback if Conductor failing

4. **Resolution** (30 min)
   - Deploy fix
   - Verify metrics return to normal
   - Document in incident log

5. **Post-Mortem** (Next day)
   - Review what happened
   - Update runbooks
   - Prevent recurrence

### Common Issues and Fixes

| Issue | Cause | Fix |
|-------|-------|-----|
| High latency | Cold starts | Pre-warm instances, enable caching |
| Memory leak | Unbounded cache | Implement LRU eviction |
| Token overrun | High complexity requests | Reduce max_tokens, batch simplify |
| D1 bottleneck | Too many connections | Use read replica, implement connection pooling |
| Failed routes | Bad conductor prompt | Update system prompt with examples |
