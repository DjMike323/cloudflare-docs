## Use Case 1: AI-Powered Customer Support

Intelligent support system that learns from tickets and improves response quality over time.

### Architecture

```
Customer → HERMES Conductor → Response Type Router
              ↓
         ┌────┼────┐
         ▼    ▼    ▼
    Urgent Inquiry FAQ   Complex
    Response Search  Analysis
         └────┼────┘
              ↓
        Unidad Dos Learning
    (ticket resolution patterns)
         ↓
    Smarter routing next time
```

### Implementation

```typescript
interface SupportTicket {
  ticketId: string;
  customerId: string;
  subject: string;
  description: string;
  priority: "low" | "medium" | "high" | "urgent";
  category?: string;
}

interface SupportResponse {
  ticketId: string;
  response: string;
  responseType: "instant" | "inquiry" | "escalation";
  confidence: number;
  resolutionTime: number;
  learnedFrom: boolean;
}

// Main support handler
async function handleSupportTicket(
  ticket: SupportTicket,
  env: Env
): Promise<SupportResponse> {
  const startTime = Date.now();

  // 1. Check if similar ticket was resolved before
  const similarTickets = await env.DB.prepare(
    `SELECT response, resolution_rating FROM episodes 
     WHERE user_id = ? AND timestamp > datetime('now', '-30 days')
     ORDER BY timestamp DESC LIMIT 5`
  )
    .bind(ticket.customerId)
    .all();

  // 2. Conductor decides response strategy
  const routing = await env.AI.run(
    "@hf/nousresearch/hermes-2-pro-mistral-7b",
    {
      messages: [
        {
          role: "system",
          content: `You are a customer support router. Analyze the ticket and decide:
- instant: Answer immediately (FAQ, common issue)
- inquiry: Ask clarifying questions
- escalation: Needs human agent

Respond with JSON: { type, confidence }`,
        },
        {
          role: "user",
          content: `Priority: ${ticket.priority}\nSubject: ${ticket.subject}\nDescription: ${ticket.description}`,
        },
      ],
      response_format: {
        type: "json_schema",
        json_schema: {
          type: "object",
          properties: {
            type: {
              type: "string",
              enum: ["instant", "inquiry", "escalation"],
            },
            confidence: { type: "number", minimum: 0, maximum: 1 },
            reasoning: { type: "string" },
          },
          required: ["type", "confidence", "reasoning"],
        },
      },
      temperature: 0.3,
    }
  );

  const decision = JSON.parse(routing.response);

  // 3. Execute based on routing decision
  let response = "";
  let responseType: "instant" | "inquiry" | "escalation" = decision.type;

  if (decision.type === "instant") {
    response = await generateInstantResponse(ticket, env);
  } else if (decision.type === "inquiry") {
    response = await generateInquiry(ticket, env);
  } else {
    response = await createEscalation(ticket, env);
  }

  // 4. Store for learning
  await env.DB.prepare(
    `INSERT INTO support_tickets (ticket_id, customer_id, subject, response, response_type, confidence, resolution_time)
     VALUES (?, ?, ?, ?, ?, ?, ?)`
  )
    .bind(
      ticket.ticketId,
      ticket.customerId,
      ticket.subject,
      response,
      responseType,
      decision.confidence,
      Date.now() - startTime
    )
    .run();

  // 5. Extract patterns for future improvements
  if (decision.confidence > 0.8 && responseType === "instant") {
    await env.DB.prepare(
      `INSERT OR REPLACE INTO semantic_memory (id, concept, definition, confidence, source)
       VALUES (?, ?, ?, ?, 'support_ticket')`
    )
      .bind(
        `pattern_${ticket.category || "general"}`,
        `${ticket.category || "support"}: ${ticket.subject.substring(0, 50)}`,
        response,
        decision.confidence
      )
      .run();
  }

  return {
    ticketId: ticket.ticketId,
    response,
    responseType,
    confidence: decision.confidence,
    resolutionTime: Date.now() - startTime,
    learnedFrom: decision.confidence > 0.75,
  };
}

async function generateInstantResponse(
  ticket: SupportTicket,
  env: Env
): Promise<string> {
  const response = await env.AI.run(
    "@hf/nousresearch/hermes-2-pro-mistral-7b",
    {
      messages: [
        {
          role: "system",
          content: "You are a helpful customer support agent. Provide a clear, concise solution.",
        },
        { role: "user", content: ticket.description },
      ],
      temperature: 0.5,
      max_tokens: 500,
    }
  );
  return response.response;
}

async function generateInquiry(ticket: SupportTicket, env: Env): Promise<string> {
  const response = await env.AI.run(
    "@hf/nousresearch/hermes-2-pro-mistral-7b",
    {
      messages: [
        {
          role: "system",
          content:
            "You are a support agent gathering more information. Ask 2-3 specific questions.",
        },
        { role: "user", content: ticket.description },
      ],
      temperature: 0.4,
      max_tokens: 300,
    }
  );
  return response.response;
}

async function createEscalation(ticket: SupportTicket, env: Env): Promise<string> {
  await env.DB.prepare(
    `INSERT INTO escalations (ticket_id, customer_id, priority, created_at)
     VALUES (?, ?, ?, datetime('now'))`
  )
    .bind(ticket.ticketId, ticket.customerId, ticket.priority)
    .run();

  return "Your ticket has been escalated to our specialist team. You'll receive a response within 2 hours.";
}
```

## Use Case 2: Content Agency Platform

Multi-client content generation with automatic quality control and brand consistency.

### Architecture

```
Client Request → Conductor (Which agent? Which client?)
      ↓
  ┌───┴───────┐
  ▼           ▼
Writer    Analyst
  ↓           ↓
Content   Quality Check
  ↓           ↓
  └─────┬─────┘
        ▓
   Unidad Dos
   (Client brand learning)
   (Quality patterns)
   (Template library)
```

### Implementation

```typescript
interface ContentRequest {
  clientId: string;
  contentType: "blog" | "email" | "social" | "description";
  topic: string;
  tone: string;
  wordCount?: number;
  userId: string;
}

interface ContentResult {
  content: string;
  qualityScore: number;
  clientFeedback: string;
  brandScore: number;
  needsRevision: boolean;
}

async function generateClientContent(
  request: ContentRequest,
  env: Env
): Promise<ContentResult> {
  // 1. Load client brand guidelines from memory
  const clientProfile = await env.DB.prepare(
    `SELECT brand_voice, tone_patterns, quality_standards FROM client_profiles WHERE client_id = ?`
  )
    .bind(request.clientId)
    .first();

  // 2. Get recent successful content for this client
  const successfulPieces = await env.DB.prepare(
    `SELECT content, quality_score FROM client_content 
     WHERE client_id = ? AND quality_score > 0.9
     ORDER BY created_at DESC LIMIT 3`
  )
    .bind(request.clientId)
    .all();

  // 3. Route to appropriate agent
  const routing = await conductorAgent(
    `Create ${request.contentType} for client with ${request.tone} tone`,
    env
  );

  // 4. Generate content with client context
  let content = "";

  if (routing.agent === "writer") {
    const systemPrompt = `You are a ${request.tone} content writer for "${clientProfile?.brand_voice || 'professional'}" brand.
${
  successfulPieces.results?.length
    ? `Recent successful examples:\n${successfulPieces.results
        .map((p: any) => p.content.substring(0, 200))
        .join("\n---\n")}`
    : ""
}`;

    const response = await env.AI.run(
      "@hf/nousresearch/hermes-2-pro-mistral-7b",
      {
        messages: [
          { role: "system", content: systemPrompt },
          {
            role: "user",
            content: `Create ${request.contentType} about "${request.topic}" (${request.wordCount || "suitable"} words)`,
          },
        ],
        temperature: 0.6,
        max_tokens: (request.wordCount || 500) + 200,
      }
    );
    content = response.response;
  }

  // 5. Quality analysis
  const qualityAnalysis = await analyzeContentQuality(content, request, env);

  // 6. Store for client learning
  await env.DB.prepare(
    `INSERT INTO client_content (client_id, content_type, content, quality_score, created_at)
     VALUES (?, ?, ?, ?, datetime('now'))`
  )
    .bind(request.clientId, request.contentType, content, qualityAnalysis.score)
    .run();

  // 7. Update brand patterns if high quality
  if (qualityAnalysis.score > 0.85) {
    await env.DB.prepare(
      `INSERT OR REPLACE INTO semantic_memory (id, concept, definition, confidence, source)
       VALUES (?, ?, ?, ?, 'client_content')`
    )
      .bind(
        `client_${request.clientId}_style`,
        `${request.tone} content pattern for ${request.contentType}`,
        content.substring(0, 500),
        qualityAnalysis.score
      )
      .run();
  }

  return {
    content,
    qualityScore: qualityAnalysis.score,
    clientFeedback: qualityAnalysis.feedback,
    brandScore: qualityAnalysis.brandAlignment,
    needsRevision: qualityAnalysis.score < 0.75,
  };
}

async function analyzeContentQuality(
  content: string,
  request: ContentRequest,
  env: Env
): Promise<{
  score: number;
  feedback: string;
  brandAlignment: number;
}> {
  const analysis = await env.AI.run(
    "@hf/nousresearch/hermes-2-pro-mistral-7b",
    {
      messages: [
        {
          role: "system",
          content: `Analyze content quality on these criteria:
- Clarity (0-1)
- Engagement (0-1)
- Brand alignment (0-1)
- SEO optimization (0-1)

Respond with JSON: { scores, feedback }`,
        },
        {
          role: "user",
          content: `Content type: ${request.contentType}\nTone: ${request.tone}\n\nContent:\n${content}`,
        },
      ],
      response_format: {
        type: "json_schema",
        json_schema: {
          type: "object",
          properties: {
            scores: { type: "object" },
            feedback: { type: "string" },
            overallScore: { type: "number" },
          },
          required: ["scores", "feedback", "overallScore"],
        },
      },
      temperature: 0.3,
    }
  );

  const result = JSON.parse(analysis.response);
  return {
    score: result.overallScore,
    feedback: result.feedback,
    brandAlignment: result.scores.brandAlignment || 0.7,
  };
}
```

## Use Case 3: Research & Analysis Platform

Autonomous research system that learns research methodologies and improves depth over time.

### Key Features

- **Automated Research**: Gather, analyze, synthesize
- **Pattern Learning**: Improve research quality over time
- **Source Tracking**: Audit trail of sources and reasoning
- **Hypothesis Testing**: Validate assumptions
- **Report Generation**: Structured output

### Core Flow

```typescript
interface ResearchRequest {
  topic: string;
  depth: "overview" | "intermediate" | "deep";
  reportType: "summary" | "analysis" | "comparison";
  userId: string;
}

async function conductResearch(
  request: ResearchRequest,
  env: Env
): Promise<{ report: string; sources: string[]; confidence: number }> {
  // 1. Plan research strategy
  const plan = await planResearch(request, env);

  // 2. Execute research phases
  const data = await env.AI.run(
    "@hf/nousresearch/hermes-2-pro-mistral-7b",
    {
      messages: [
        {
          role: "system",
          content: `You are a research analyst. ${plan.strategy}
Research phases: ${plan.phases.join(", ")}`,
        },
        { role: "user", content: `Research topic: ${request.topic}` },
      ],
      temperature: 0.4,
      max_tokens: 2000,
    }
  );

  // 3. Analyze and synthesize
  const synthesis = await synthesizeFindings(data.response, request, env);

  // 4. Generate structured report
  const report = await generateResearchReport(synthesis, request, env);

  // 5. Store research methodology for learning
  await storeResearchLearning(request, synthesis, env);

  return {
    report,
    sources: synthesis.sources,
    confidence: synthesis.confidence,
  };
}

async function planResearch(request: ResearchRequest, env: Env): Promise<any> {
  // Load previous successful research plans
  const similar = await env.DB.prepare(
    `SELECT strategy, phases FROM research_plans 
     WHERE topic LIKE ? ORDER BY quality_score DESC LIMIT 1`
  )
    .bind(`%${request.topic.split(" ")[0]}%`)
    .first();

  return {
    strategy: similar?.strategy || "Use systematic approach",
    phases: similar?.phases || ["define", "research", "analyze", "synthesize"],
  };
}
```

Automated product catalog optimization with dynamic descriptions and recommendations.

- **SEO Optimization**: Auto-optimize for search
- **Pricing Intelligence**: Suggest pricing based on patterns

### Implementation Highlights

```typescript
async function optimizeProduct(
  productId: string,
  category: string,
  env: Env
): Promise<{
  description: string;
  keywords: string[];
```
